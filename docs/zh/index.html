1 function twoSum(nums: number[], target: number) {
2   const map = new Map<number, number>();
3
4   for (let i = 0; i < nums.length; i++) {
5     const complement = target - nums[i];
6     if (map.has(complement)) {
7       return [map.get(complement)!, i];
8     }
9     map.set(nums[i], i);
10  }
11  return [];
12 }

可以解释一下这个解法的时间复杂度吗？

这个解法在遍历时使用哈希表保存数字。

  • 时间复杂度：O(n)
    只需遍历数组一次。
  • 空间复杂度：O(n)
    最多保存 n 个元素。
